cmake_minimum_required(VERSION 3.19)
project(dnr_laufey LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
  enable_language(OBJCXX)
endif()
if(NOT DNR_BACKEND MATCHES "^(webview|system-cef|dual)$")
  message(FATAL_ERROR "Unknown DNR_BACKEND: ${DNR_BACKEND}")
endif()
add_subdirectory(${LAUFEY_SOURCE_DIR}/backend-common backend-common)
add_library(dnr_bridge OBJECT bridge.cc)
add_library(dnr_laufey STATIC $<TARGET_OBJECTS:dnr_bridge>)
target_link_libraries(dnr_laufey PRIVATE laufey_backend_common)

if(DNR_BACKEND STREQUAL "system-cef" OR DNR_BACKEND STREQUAL "dual")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "system-cef and dual require native Linux x86_64")
  endif()
  list(APPEND CMAKE_MODULE_PATH "/usr/share/cmake/Modules")
  set(CEF_API_VERSION 14900 CACHE STRING "Laufey 0.7.0 CEF API")
  find_package(CEF REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CEF_NATIVE REQUIRED gtk+-3.0 xi x11)
  set(SOURCE ${LAUFEY_SOURCE_DIR}/cef/src)
  add_library(dnr_cef OBJECT
    ${SOURCE}/app.cc ${SOURCE}/scheme_handler.cc ${SOURCE}/runtime_loader.cc
    ${SOURCE}/runtime_loader_linux.cc ${SOURCE}/render_process_handler.cc
    ${SOURCE}/renderer_app.cc ${SOURCE}/main_linux.cc)
  target_compile_definitions(dnr_cef PRIVATE CEF_API_VERSION=14900
    RuntimeLoader=DnrCefRuntimeLoader LaufeyFindColocatedRuntime=DnrCefFindColocatedRuntime)
  set_source_files_properties(${SOURCE}/main_linux.cc PROPERTIES COMPILE_DEFINITIONS "main=dnr_cef_main")
  target_compile_options(dnr_cef PRIVATE -fno-stack-protector ${CEF_NATIVE_CFLAGS_OTHER})
  target_include_directories(dnr_cef PRIVATE ${CEF_NATIVE_INCLUDE_DIRS} ${SOURCE}
    ${LAUFEY_SOURCE_DIR}/capi/include ${LAUFEY_SOURCE_DIR}/backend-common/include)
  target_link_libraries(dnr_cef PRIVATE CEF::Wrapper CEF::Library)
  target_sources(dnr_laufey PRIVATE $<TARGET_OBJECTS:dnr_cef>)
  target_compile_definitions(dnr_bridge PRIVATE DNR_SYSTEM_CEF=1 CEF_API_VERSION=14900)
  target_link_libraries(dnr_bridge PRIVATE CEF::Wrapper CEF::Library)
  target_link_libraries(dnr_laufey PRIVATE CEF::Wrapper CEF::Library)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cef-wrapper-path" CONTENT "$<TARGET_FILE:CEF::Wrapper>")
endif()

if(DNR_BACKEND STREQUAL "webview" OR DNR_BACKEND STREQUAL "dual")
  set(SOURCE ${LAUFEY_SOURCE_DIR}/webview/src)
  add_library(dnr_webview OBJECT ${SOURCE}/runtime_loader.cc)
  target_compile_definitions(dnr_webview PRIVATE
    RuntimeLoader=DnrWebViewRuntimeLoader LaufeyFindColocatedRuntime=DnrWebViewFindColocatedRuntime)
  if(APPLE)
    target_sources(dnr_webview PRIVATE ${SOURCE}/webview_macos.mm ${SOURCE}/main_mac.mm)
    target_compile_options(dnr_webview PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBVIEW_NATIVE REQUIRED webkit2gtk-4.1 gtk+-3.0)
    target_sources(dnr_webview PRIVATE ${SOURCE}/webview_linux.cc ${SOURCE}/main_linux.cc)
    target_include_directories(dnr_webview PRIVATE ${WEBVIEW_NATIVE_INCLUDE_DIRS})
    target_compile_options(dnr_webview PRIVATE ${WEBVIEW_NATIVE_CFLAGS_OTHER})
  endif()
  set_source_files_properties(${SOURCE}/main_mac.mm ${SOURCE}/main_linux.cc PROPERTIES COMPILE_DEFINITIONS "main=dnr_webview_main")
  target_include_directories(dnr_webview PRIVATE ${SOURCE}
    ${LAUFEY_SOURCE_DIR}/capi/include ${LAUFEY_SOURCE_DIR}/backend-common/include)
  target_sources(dnr_laufey PRIVATE $<TARGET_OBJECTS:dnr_webview>)
  target_compile_definitions(dnr_bridge PRIVATE DNR_WEBVIEW=1)
endif()
install(TARGETS dnr_laufey laufey_backend_common ARCHIVE DESTINATION lib)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "GUI import trampolines require the x86_64 SysV ABI")
  endif()
  enable_language(ASM)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(DNR_READELF readelf REQUIRED)
  set(import_manifest "common|$<TARGET_FILE:laufey_backend_common>\ncef|$<JOIN:$<TARGET_OBJECTS:dnr_bridge>,\ncef|>\n")
  set(import_dependencies dnr_laufey laufey_backend_common)
  # backend-common's pkg-config result is scoped to its subdirectory; GTK is
  # already included in either backend's native library list above.
  set(import_libraries ${CEF_NATIVE_LIBRARIES} ${WEBVIEW_NATIVE_LIBRARIES})
  list(REMOVE_DUPLICATES import_libraries)
  foreach(library IN LISTS import_libraries)
    if(library MATCHES "^(m|pthread|dl)$")
      continue()
    endif()
    unset(import_library CACHE)
    find_library(import_library NAMES ${library}
      HINTS ${CEF_NATIVE_LIBRARY_DIRS} ${WEBVIEW_NATIVE_LIBRARY_DIRS} REQUIRED)
    string(APPEND import_manifest "library|${import_library}\n")
    list(APPEND import_dependencies "${import_library}")
  endforeach()
  if(TARGET dnr_cef)
    string(APPEND import_manifest "cef|$<JOIN:$<TARGET_OBJECTS:dnr_cef>,\ncef|>\ncef|$<TARGET_FILE:CEF::Wrapper>\nlibrary|$<TARGET_FILE:CEF::Library>\n")
    list(APPEND import_dependencies dnr_cef CEF::Wrapper "$<TARGET_FILE:CEF::Library>")
  endif()
  if(TARGET dnr_webview)
    string(APPEND import_manifest "webview|$<JOIN:$<TARGET_OBJECTS:dnr_webview>,\nwebview|>\n")
    list(APPEND import_dependencies dnr_webview)
  endif()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/gui-inputs" CONTENT "${import_manifest}")
  add_custom_command(
    OUTPUT "${CMAKE_BINARY_DIR}/gui_trampolines.S" "${CMAKE_BINARY_DIR}/gui_imports.inc"
           "${CMAKE_BINARY_DIR}/gui-wrap-flags" "${CMAKE_BINARY_DIR}/gui-imports.json"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/generate_gui_imports.py"
      --manifest "${CMAKE_BINARY_DIR}/gui-inputs" --output "${CMAKE_BINARY_DIR}"
      --readelf "${DNR_READELF}"
    DEPENDS ${import_dependencies} "${CMAKE_BINARY_DIR}/gui-inputs" generate_gui_imports.py
    VERBATIM)
  add_library(dnr_gui_imports STATIC gui_loader.cc "${CMAKE_BINARY_DIR}/gui_trampolines.S"
    "${CMAKE_BINARY_DIR}/gui_imports.inc")
  target_include_directories(dnr_gui_imports PRIVATE "${CMAKE_BINARY_DIR}")
  target_compile_definitions(dnr_bridge PRIVATE DNR_LAZY_GUI=1)
  install(TARGETS dnr_gui_imports ARCHIVE DESTINATION lib)
endif()
