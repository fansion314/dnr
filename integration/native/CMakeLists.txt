cmake_minimum_required(VERSION 3.19)
project(dnr_laufey LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
  enable_language(OBJCXX)
endif()
add_subdirectory(${LAUFEY_SOURCE_DIR}/backend-common backend-common)
if(DNR_BACKEND STREQUAL "system-cef")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "system-cef requires native Linux x86_64")
  endif()
  list(APPEND CMAKE_MODULE_PATH "/usr/share/cmake/Modules")
  set(CEF_API_VERSION 14900 CACHE STRING "Laufey 0.7.0 CEF API")
  find_package(CEF REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(NATIVE REQUIRED gtk+-3.0 xi x11)
  set(SOURCE ${LAUFEY_SOURCE_DIR}/cef/src)
  add_library(dnr_laufey STATIC bridge.cc
    ${SOURCE}/app.cc ${SOURCE}/scheme_handler.cc ${SOURCE}/runtime_loader.cc
    ${SOURCE}/runtime_loader_linux.cc ${SOURCE}/render_process_handler.cc
    ${SOURCE}/renderer_app.cc ${SOURCE}/main_linux.cc)
  target_compile_definitions(dnr_laufey PRIVATE DNR_SYSTEM_CEF=1 CEF_API_VERSION=14900)
  target_compile_options(dnr_laufey PRIVATE -fno-stack-protector ${NATIVE_CFLAGS_OTHER})
  target_include_directories(dnr_laufey PRIVATE ${NATIVE_INCLUDE_DIRS})
  target_link_libraries(dnr_laufey PRIVATE CEF::Wrapper CEF::Library)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cef-wrapper-path" CONTENT "$<TARGET_FILE:CEF::Wrapper>")
else()
  set(SOURCE ${LAUFEY_SOURCE_DIR}/webview/src)
  add_library(dnr_laufey STATIC bridge.cc ${SOURCE}/runtime_loader.cc)
  if(APPLE)
    target_sources(dnr_laufey PRIVATE ${SOURCE}/webview_macos.mm ${SOURCE}/main_mac.mm)
    target_compile_options(dnr_laufey PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NATIVE REQUIRED webkit2gtk-4.1 gtk+-3.0)
    target_sources(dnr_laufey PRIVATE ${SOURCE}/webview_linux.cc ${SOURCE}/main_linux.cc)
    target_include_directories(dnr_laufey PRIVATE ${NATIVE_INCLUDE_DIRS})
    target_compile_options(dnr_laufey PRIVATE ${NATIVE_CFLAGS_OTHER})
  endif()
endif()
set_source_files_properties(${SOURCE}/main_mac.mm ${SOURCE}/main_linux.cc PROPERTIES COMPILE_DEFINITIONS "main=dnr_backend_main_impl")
target_include_directories(dnr_laufey PRIVATE ${SOURCE} ${LAUFEY_SOURCE_DIR}/capi/include ${LAUFEY_SOURCE_DIR}/backend-common/include)
target_link_libraries(dnr_laufey PRIVATE laufey_backend_common)
install(TARGETS dnr_laufey laufey_backend_common ARCHIVE DESTINATION lib)
